cmake_minimum_required(VERSION 3.25)
project(LumyteSetupSmoke LANGUAGES CXX)
if(WIN32)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    find_package(VulkanHeaders CONFIG REQUIRED)
    find_package(VulkanLoader CONFIG REQUIRED)
else()
    find_package(Vulkan REQUIRED)
endif()
find_package(ZLIB REQUIRED)
add_library(lumyte_setup_smoke SHARED native.cpp)
set_property(TARGET lumyte_setup_smoke PROPERTY COMPILE_WARNING_AS_ERROR ON)
target_compile_features(lumyte_setup_smoke PRIVATE cxx_std_20)
target_link_libraries(lumyte_setup_smoke PRIVATE ZLIB::ZLIB)
if(WIN32)
    target_link_libraries(lumyte_setup_smoke PRIVATE Vulkan::Loader Vulkan::Headers d3d12 dxgi)
else()
    target_link_libraries(lumyte_setup_smoke PRIVATE Vulkan::Vulkan)
endif()
