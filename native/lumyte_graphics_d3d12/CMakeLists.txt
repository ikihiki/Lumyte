cmake_minimum_required(VERSION 3.25)
project(lumyte_graphics_d3d12 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Direct3D 12 requires Windows")
endif()
add_library(lumyte_graphics_d3d12 SHARED d3d12.cpp)
if(NOT EXISTS "${AGILITY_SDK_ROOT}/build/native/include/d3d12.h")
  message(FATAL_ERROR "AGILITY_SDK_ROOT must point to the restored Agility SDK package")
endif()
if(AGILITY_CHANNEL STREQUAL "Preview")
  target_compile_definitions(lumyte_graphics_d3d12 PRIVATE LUMYTE_AGILITY_VERSION=D3D12_PREVIEW_SDK_VERSION LUMYTE_PREVIEW=1)
elseif(AGILITY_CHANNEL STREQUAL "Stable")
  target_compile_definitions(lumyte_graphics_d3d12 PRIVATE LUMYTE_AGILITY_VERSION=D3D12_SDK_VERSION)
else()
  message(FATAL_ERROR "AGILITY_CHANNEL must be Stable or Preview")
endif()
target_include_directories(lumyte_graphics_d3d12 BEFORE PRIVATE "${AGILITY_SDK_ROOT}/build/native/include")
target_compile_features(lumyte_graphics_d3d12 PRIVATE cxx_std_20)
target_compile_definitions(lumyte_graphics_d3d12 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(lumyte_graphics_d3d12 PRIVATE /W4 /WX /EHsc)
target_link_libraries(lumyte_graphics_d3d12 PRIVATE d3d12 dxgi user32)
